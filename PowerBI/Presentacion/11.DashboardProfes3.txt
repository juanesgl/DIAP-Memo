var nc=document.getElementById('n1'),nx=nc.getContext('2d'),nw,nh,nd=[];
var pcols=['106,27,154','26,101,164','211,47,47','16,157,125'];

function ni(){
  nw=nc.width=nc.parentElement.offsetWidth;
  nh=nc.height=nc.parentElement.offsetHeight;
  nd=Array.from({length:20},function(){
    return {
      x:Math.random()*nw,
      y:Math.random()*nh,
      vx:(Math.random()-.5)*.25,
      vy:(Math.random()-.5)*.25,
      r:Math.random()*2+1,
      co:pcols[Math.floor(Math.random()*pcols.length)]
    };
  });
}

function ndr(){
  nx.clearRect(0,0,nw,nh);
  nd.forEach(function(n){
    n.x+=n.vx; n.y+=n.vy;
    if(n.x<0||n.x>nw)n.vx*=-1;
    if(n.y<0||n.y>nh)n.vy*=-1;
  });
  
  nd.forEach(function(a,i){
    nd.forEach(function(b,j){
      if(j<=i)return;
      var d=Math.hypot(a.x-b.x,a.y-b.y);
      if(d<180){
        nx.strokeStyle='rgba('+a.co+','+(0.15*(1-d/180))+')';
        nx.lineWidth=.6;
        nx.beginPath();
        nx.moveTo(a.x,a.y);
        nx.lineTo(b.x,b.y);
        nx.stroke();
      }
    });
  });

  nd.forEach(function(n){
    nx.beginPath();
    nx.arc(n.x,n.y,n.r,0,Math.PI*2);
    nx.fillStyle='rgba('+n.co+',0.5)';
    nx.fill();
  });

  requestAnimationFrame(ndr);
}

ni();
ndr();
window.addEventListener('resize',ni);
</script>
